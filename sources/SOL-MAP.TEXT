function MapVal (c : char) : integer;
{ One run length: printable ASCII from ! upward, skipping the quote, the }
{ parentheses, the star and the braces.                                  }
var k : integer;
begin
   k := ord(c) - 33;
   if ord(c) > 42 then k := k - 4;
   if ord(c) > 123 then k := k - 1;
   if ord(c) > 125 then k := k - 1;
   MapVal := k
end;

procedure SmTables;
{ Lead (pixel i to the byte's end) and trail (byte start to pixel i)   }
{ bits of black and of the two gray rows (170: even pixels, 85: odd).  }
var t, i, v, bitv : integer;
begin
   for t := 0 to 2 do begin
      v := 0;
      bitv := 128;
      for i := 0 to 7 do begin
         if (t = 0) or ((i mod 2) = t - 1) then v := v + bitv;
         smTrl[t, i] := v;
         bitv := bitv div 2
      end;
      smFull[t] := v;
      v := 0;
      bitv := 1;
      for i := 7 downto 0 do begin
         if (t = 0) or ((i mod 2) = t - 1) then v := v + bitv;
         smLead[t, i] := v;
         bitv := bitv * 2
      end
   end
end;

procedure SmOr (i, add : integer);
{ Byte i gets add's bits (they are never set already). }
var v : integer;
begin
   v := sb.b[i];
   if v < 0 then v := v + 256;
   v := v + add;
   if v > 127 then v := v - 256;
   sb.b[i] := v
end;

procedure SmRun (base, xa, xb, t : integer);
{ Pixels xa..xb of the band row at byte base, in black (t 0) or gray. }
var fb, lb, k, f : integer;
begin
   if xa < 0 then xa := 0;
   if xb > smW - 1 then xb := smW - 1;
   if xa > xb then exit(SmRun);
   fb := xa div 8;
   lb := xb div 8;
   if fb = lb then
      SmOr(base + fb, smLead[t, xa mod 8] + smTrl[t, xb mod 8] - smFull[t])
   else begin
      SmOr(base + fb, smLead[t, xa mod 8]);
      f := smFull[t];
      if f > 127 then f := f - 256;
      for k := fb + 1 to lb - 1 do
         sb.b[base + k] := f;
      SmOr(base + lb, smTrl[t, xb mod 8])
   end
end;

function SmAcos (c : real) : real;
{ arccos(c) in degrees, -1 < c < 1, from SmAcT (64 steps, linear between): }
{ no trig on the Lisa (REV B: 6 calls a row made the map take 30 s).        }
var u : real;
    i : integer;
    a : real;
begin
   u := (c + 1.0) * 32.0;
   i := trunc(u);
   if i > 63 then i := 63;
   if i < 0 then i := 0;
   a := SmAcT(i);
   SmAcos := a + (SmAcT(i + 1) - a) * (u - i)
end;

procedure SmNight (y : integer; tanDec, subLon : real);
{ The night columns of map row y: smNn runs smNa[i]..smNb[i]. }
var ch, h0, xc, hw : real;
    a, b           : integer;
begin
   ch := -SmTanL(y) * tanDec;
   smNn := 0;
   if ch >= 1.0 then begin                      { the Sun does not rise here today }
      smNn := 1;
      smNa[1] := 0;
      smNb[1] := smW - 1
   end
   else if ch > -1.0 then begin
      h0 := SmAcos(ch);                          { the half-day, degrees }
      xc := (subLon + 180.0) * smW / 360.0;
      hw := h0 * smW / 360.0;
      a := round(xc - hw);
      b := round(xc + hw);
      if (a >= 0) and (b <= smW - 1) then begin
         smNn := 2;
         smNa[1] := 0;      smNb[1] := a - 1;
         smNa[2] := b + 1;  smNb[2] := smW - 1
      end
      else if a < 0 then begin
         smNn := 1;
         smNa[1] := b + 1;  smNb[1] := a + smW - 1
      end
      else begin
         smNn := 1;
         smNa[1] := b - smW + 1;  smNb[1] := a - 1
      end
   end
end;

procedure SmSea (base, xa, xb, par : integer);
{ A sea run: gray where it is night. }
var i, lo, hi : integer;
begin
   for i := 1 to smNn do begin
      lo := xa;
      if smNa[i] > lo then lo := smNa[i];
      hi := xb;
      if smNb[i] < hi then hi := smNb[i];
      if lo <= hi then SmRun(base, lo, hi, 1 + par)
   end
end;

procedure SmDraw {left, top : integer; dec, subLon, tLat, tLon : real};
var band, y0, y1, y, x, i, v, par, base : integer;
    sx, sy                              : integer;
    s                                   : Str255;
    onLand                              : boolean;
    bm                                  : BitMap;
    sr, dr, vb, r                       : Rect;
    tanDec, dr2                         : real;
begin
   SmTables;
   dr2 := dec * smPi / 180.0;
   tanDec := sin(dr2) / cos(dr2);                { once a drawing, not once a row }
   vb := thePort^.visRgn^^.rgnBBox;
   bm.baseAddr := @sb;
   bm.rowBytes := smRb;
   for band := 0 to (smH - 1) div smBandH do begin
      y0 := band * smBandH;
      y1 := y0 + smBandH - 1;
      if y1 > smH - 1 then
         y1 := smH - 1;
      if (top + y1 >= vb.top) and (top + y0 < vb.bottom) then begin
         for i := 0 to (smRb * smBandH) div 2 - 1 do
            sb.w[i] := 0;
         for y := y0 to y1 do begin
            MapRow(y, s);
            SmNight(y, tanDec, subLon);
            if odd(top + y) then par := 1 else par := 0;
            base := (y - y0) * smRb;
            x := 0;
            onLand := false;
            for i := 1 to length(s) do begin
               v := MapVal(s[i]);
               if v > 0 then begin
                  if onLand then
                     SmRun(base, x, x + v - 1, 0)
                  else
                     SmSea(base, x, x + v - 1, par)
               end;
               x := x + v;
               onLand := not onLand
            end;
            SmSea(base, x, smW - 1, par)       { the sea after the last run }
         end;
         SetRect(bm.bounds, 0, 0, smW, y1 - y0 + 1);
         SetRect(sr, 0, 0, smW, y1 - y0 + 1);
         SetRect(dr, left, top + y0, left + smW, top + y1 + 1);
         CopyBits(bm, thePort^.portBits, sr, dr, srcCopy, nil)
      end
   end;
   PenNormal;
   SetRect(r, left - 1, top - 1, left + smW + 1, top + smH + 1);
   FrameRect(r);
   { the Sun's overhead point: a black dot in a white ring }
   sx := left + round((subLon + 180.0) * smW / 360.0);
   sy := top + round((80.0 - dec) * smH / 140.0);
   SetRect(r, sx - 7, sy - 5, sx + 8, sy + 6);
   EraseOval(r);
   FrameOval(r);
   SetRect(r, sx - 3, sy - 2, sx + 4, sy + 3);
   PaintOval(r);
   { your town: a cross on a white patch }
   if (tLat < 79.0) and (tLat > -59.0) then begin
      sx := left + round((tLon + 180.0) * smW / 360.0);
      sy := top + round((80.0 - tLat) * smH / 140.0);
      SetRect(r, sx - 6, sy - 4, sx + 7, sy + 5);
      EraseRect(r);
      SetRect(r, sx - 5, sy, sx + 6, sy + 1);
      PaintRect(r);
      SetRect(r, sx - 1, sy - 3, sx + 1, sy + 4);
      PaintRect(r)
   end
end;
