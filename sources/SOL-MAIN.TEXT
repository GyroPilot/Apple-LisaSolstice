procedure Clip (var big : Str255; var out : StatStr);
{ Copy at most 80 characters.  HARDWARE LESSON (INS REV B): COPY(big,   }
{ 1, 80) with LENGTH(big) < 80 returned an EMPTY string on the Lisa.    }
var n : integer;
begin
   n := length(big);
   if n > 80 then n := 80;
   if n = 0 then
      out := ''
   else
      out := copy(big, 1, n)
end;

procedure DrawShow;
{ Repaint the whole status pane; under the lines, the orbit or the tilt }
{ picture when one is showing (LisaSolstice).                           }
var fi : FontInfo;
    i  : integer;
    r  : Rect;
begin
   if myWindow = nil then
      exit(DrawShow);
   SetPort(myWindow);
   ClipRect(myWindow^.portRect);        { FolderSize left the clip at the old size (Checkers SyncPort) }
   EraseRect(myWindow^.portRect);
   GetFontInfo(fi);
   for i := 1 to statCnt do begin
      MoveTo(10, 6 + (i - 1) * lineHt + fi.ascent);
      DrawString(statLine[i])
   end;
   if pageMode = pgSplash then
      SpDraw(10, 8 + statCnt * lineHt)
   else if pageMode = pgPhoto then begin
      { REV B: a picture from the talk, centred under its two lines }
      r := myWindow^.portRect;
      i := (r.right - r.left - PhW(phNo)) div 2;
      if i < 10 then i := 10;
      PhDraw(phNo, i, 10 + statCnt * lineHt)
   end
   else if pageMode = pgTime then begin
      r := myWindow^.portRect;
      r.top := r.top + 10 + statCnt * lineHt;
      r.left := r.left + 6;
      r.right := r.right - 6;
      r.bottom := r.bottom - 6;
      VwTime(tlYear, r)
   end
   else if pageMode = pgMap then begin
      { day and night now (from Solstice for Mac) }
      i := 12 + statCnt * lineHt;
      SaSun(pgDay, pgFr);
      SmDraw(74, i, saDecl, saSubLn, vwLat, vwLon);
      MoveTo(10, i + smH + 20);
      DrawString(mpL1);
      MoveTo(10, i + smH + 20 + lineHt);
      DrawString(mpL2)
   end
   else if pageMode <> pgText then begin
      r := myWindow^.portRect;
      r.top := r.top + 10 + statCnt * lineHt;
      r.left := r.left + 6;
      r.right := r.right - 6;
      r.bottom := r.bottom - 6;
      if pageMode = pgOrbit then
         VwOrbit(pgDay, pgFr, r)
      else
         VwTilt(pgDay, pgFr, r)
   end
end;

procedure AddLine (var s : StatStr);
{ Append a line; scroll the list up when full; repaint. }
var i : integer;
begin
   if statCnt < maxLines then
      statCnt := statCnt + 1
   else
      for i := 1 to maxLines - 1 do
         statLine[i] := statLine[i + 1];
   statLine[statCnt] := s;
   if not quiet then DrawShow
end;

procedure SetLast (var s : StatStr);
{ Replace the last line and repaint that line only (REV E). }
begin
   if statCnt = 0 then
      AddLine(s)
   else begin
      statLine[statCnt] := s;
      DrawOne(statCnt)
   end
end;

procedure SayDate (var lead : StatStr; y, doy, h, mi : integer);
{ One line: lead, then "Saturday, October 10, 1987   14:05".  y is the  }
{ year whose calendar is used AND the year printed; doy is the day of   }
{ that year.                                                             }
var m, d, wd : integer;
    part     : Str255;
    mins     : Str255;
    big      : Str255;
begin
   TwMonDay(y, doy, m, d);
   wd := TwWkDay(y, m, d);
   TwDayNm(wd, part);
   big := concat(lead, part);
   TwMonNm(m, part);
   big := concat(big, concat(', ', part));
   TwNum(d, part);
   big := concat(big, concat(' ', part));
   TwNum(y, part);
   big := concat(big, concat(', ', part));
   TwNum(h, part);
   Tw2Dig(mi, mins);
   big := concat(big, concat('   ', part));
   big := concat(big, concat(':', mins));
   AddBig(big)
end;

procedure NetSrcNm (var big : Str255);
{ The source in use and the speed, in words. }
var part : Str255;
begin
   if ntSrc = ntSTown then begin
      if ntHave then
         big := concat('the web (timeapi.io) for ', ntTown)
      else
         big := 'the web for your town (no town yet: T in LisaFuji)'
   end
   else if ntSrc = ntSZone then begin
      if length(ntZone) > 0 then
         big := concat('the web (timeapi.io) for ', ntZone)
      else
         big := 'the web for a time zone (none typed yet)'
   end
   else if fnVia = fnViaMb then         { REV H: the ESPFuji drive's clock (SNTP; timezone= in espfuji.ini) }
      big := 'the ESProFile''s clock (ESPFuji; its time zone is set in espfuji.ini)'
   else
      big := 'the FujiNet''s clock (its time zone is set on the FujiNet)';
   if fnVia <> fnViaMb then begin        { REV H: no baud rate on the drive }
      TwNum(ntBaud, part);
      big := concat(big, concat(', ', concat(part, ' baud')))
   end
end;

procedure ShowClock;
var y, doy, h, mi, s : integer;
    ry, rm, rd, rt   : integer;
    m, d             : integer;
    haveNote         : boolean;
    row              : StatStr;
    lead             : StatStr;
    part             : Str255;
begin
   promptOn := false;
   promptStep := esNone;
   ckMode := 0;
   LogMode;
   quiet := true;
   row := solVersion;
   AddLine(row);
   row := '';
   AddLine(row);
   haveNote := TwLoad(myVol, ry, rm, rd, rt);
   if TwGet(y, doy, h, mi, s) then begin
      lead := 'Lisa clock:  ';
      SayDate(lead, y, doy, h, mi);
      if haveNote and (rt = y) and (ry <> y) then begin
         TwMonDay(y, doy, m, d);
         lead := 'Real date:   ';
         SayDate(lead, ry, TwDayOY(ry, m, d), h, mi)
      end
      else begin
         row := '(No note of the real year for this clock setting.)';
         AddLine(row)
      end
   end
   else begin
      row := 'The Lisa clock is NOT SET: it lost power and reads 1980.';
      AddLine(row);
      row := 'Press S to set it.';
      AddLine(row)
   end;
   row := '';
   AddLine(row);
   NetSrcNm(part);
   part := concat('Network: ', part);
   AddBig(part);
   row := '';
   AddLine(row);
   row := 'S = set by hand    N = set from the network    C = show    H = help';
   AddLine(row);
   quiet := false;
   DrawShow
end;

procedure LoadDefs;
{ The values offered in brackets.  The note file gives the real year    }
{ and the last date; a clock that is still running gives a better       }
{ month, day, hour and minute.  -1 = nothing to offer.                  }
var ry, rm, rd, rt   : integer;
    y, doy, h, mi, s : integer;
    m, d             : integer;
begin
   dfYear  := baseYear;
   dfMonth := -1;
   dfDay   := -1;
   dfHour  := -1;
   dfMin   := -1;
   if TwLoad(myVol, ry, rm, rd, rt) then begin
      dfYear  := ry;
      dfMonth := rm;
      dfDay   := rd
   end;
   if TwGet(y, doy, h, mi, s) then begin
      TwMonDay(y, doy, m, d);
      dfMonth := m;
      dfDay   := d;
      dfHour  := h;
      dfMin   := mi
   end
end;

procedure AskStep;
{ Put up the prompt for promptStep on a new last line. }
var what : StatStr;
    dflt : integer;
    part : Str255;
    big  : Str255;
    row  : StatStr;
begin
   what := 'Minute (0-59)';
   dflt := dfMin;
   if promptStep = esYear then begin
      what := 'Year';
      dflt := dfYear
   end
   else if promptStep = esMonth then begin
      what := 'Month (1-12)';
      dflt := dfMonth
   end
   else if promptStep = esDay then begin
      what := 'Day';
      dflt := dfDay
   end
   else if promptStep = esHour then begin
      what := 'Hour (0-23)';
      dflt := dfHour
   end;
   if dflt >= 0 then begin
      TwNum(dflt, part);
      big := concat(what, concat(' [', part));
      big := concat(big, ']: ')
   end
   else
      big := concat(what, ': ');
   Clip(big, promptLbl);
   promptTxt := '';
   promptOn := true;
   row := '';
   AddLine(row);
   DrawPrompt
end;

procedure Confirm;
{ All five values are in.  Show both dates, ask, set, read back, save   }
{ the note.                                                              }
var twinY : integer;
    doy   : integer;
    rc    : integer;
    exact : boolean;
    go    : boolean;
    row   : StatStr;
    lead  : StatStr;
    part  : Str255;
    big   : Str255;
begin
   twinY := TwYear(inYear, exact);
   if (inMonth = 2) and (inDay = 29) and (not TwLeap(twinY)) then
      doy := TwDayOY(twinY, 3, 1)       { defensive: cannot happen }
   else
      doy := TwDayOY(twinY, inMonth, inDay);

   row := '';
   AddLine(row);
   lead := 'Today:               ';
   SayDate(lead, inYear, TwDayOY(inYear, inMonth, inDay), inHour, inMin);
   lead := 'Lisa will be set to: ';
   SayDate(lead, twinY, doy, inHour, inMin);

   if exact then
      go := CautionAlert(allAlerts, alrtSetGo)
   else begin
      TwNum(inYear, part);
      big := concat(part, ' has no twin year: dates right, weekday names wrong.');
      AddBig(big);
      go := CautionAlert(allAlerts, alrtNoTwin)
   end;
   if not go then begin
      row := 'Cancelled.  The clock was not changed.';
      AddLine(row);
      exit(Confirm)
   end;

   rc := TwSet(twinY, doy, inHour, inMin);
   if rc = 0 then begin
      row := 'The clock is set.  Press C to see it.';
      AddLine(row);
      if not TwSave(myVol, inYear, inMonth, inDay, twinY) then begin
         TwNum(twEcode, part);
         big := concat('(The note file was not saved: ecode ', part);
         big := concat(big, '.)');
         AddBig(big)
      end
   end
   else begin
      TwNum(rc, part);
      big := concat('The clock was NOT set (result ', part);
      big := concat(big, ').');
      AddBig(big);
      if rc = 2 then
         row := 'The read-back did not match; the old time was put back.'
      else if rc = 3 then
         row := 'The read-back did not match; there was no old time to restore.'
      else
         row := 'A value was out of range; nothing was written.';
      AddLine(row);
      NoteAlert(allAlerts, alrtFail)
   end
end;

procedure StepDone;
{ RETURN at a prompt.  Empty = take the default.  A value out of range  }
{ says so and asks again.  After the minute, Confirm.                    }
var n, i   : integer;
    lo, hi : integer;
    dflt   : integer;
    row    : StatStr;
    part   : Str255;
    big    : Str255;
begin
   promptOn := false;

   dflt := dfMin;
   lo := 0;
   hi := 59;
   if promptStep = esYear then begin
      dflt := dfYear;
      lo := twLoReal;
      hi := twHiReal
   end
   else if promptStep = esMonth then begin
      dflt := dfMonth;
      lo := 1;
      hi := 12
   end
   else if promptStep = esDay then begin
      dflt := dfDay;
      lo := 1;
      hi := TwDaysIn(inYear, inMonth)
   end
   else if promptStep = esHour then begin
      dflt := dfHour;
      lo := 0;
      hi := 23
   end;

   if length(promptTxt) = 0 then
      n := dflt
   else begin
      n := 0;
      for i := 1 to length(promptTxt) do
         n := n * 10 + (ord(promptTxt[i]) - ord('0'))
   end;

   if (n < lo) or (n > hi) then begin
      TwNum(lo, part);
      big := concat('   That needs a number from ', part);
      TwNum(hi, part);
      big := concat(big, concat(' to ', part));
      big := concat(big, '.');
      Clip(big, row);
      SetLast(row);
      AskStep;
      exit(StepDone)
   end;

   { accepted: the prompt line becomes the settled value }
   TwNum(n, part);
   big := concat(promptLbl, part);
   Clip(big, row);
   SetLast(row);

   if promptStep = esYear then
      inYear := n
   else if promptStep = esMonth then
      inMonth := n
   else if promptStep = esDay then
      inDay := n
   else if promptStep = esHour then
      inHour := n
   else
      inMin := n;

   if promptStep < esMin then begin
      promptStep := promptStep + 1;
      AskStep
   end
   else begin
      promptStep := esNone;
      Confirm
   end
end;

procedure NetAsk;
{ N: the time from the network, both dates, and the Y question. }
var rc                 : integer;
    y, mo, d, h, mi, s : integer;
    exact              : boolean;
    row                : StatStr;
    lead               : StatStr;
    part               : Str255;
    big                : Str255;
begin
   promptOn := false;
   promptStep := esNone;
   ckMode := 0;
   LogMode;
   row := 'Set the clock from the network';
   AddLine(row);
   NetSrcNm(part);
   big := concat('From ', part);
   AddBig(big);
   row := '';
   AddLine(row);
   row := 'Asking the FujiNet ...';
   AddLine(row);
   rc := NtGet(y, mo, d, h, mi, s);
   if rc = 4 then begin
      AskZone;
      exit(NetAsk)
   end;
   if rc = 1 then begin
      row := 'Serial B would not open.  Quit LISACom or LisaFuji if one is open,';
      SetLast(row);
      row := 'or set the Serial B Connector to Serial Cable (Preferences).';
      AddLine(row);
      exit(NetAsk)
   end;
   if rc = 3 then begin
      row := 'No town yet.  Set one in LisaFuji (T), on this disk, or choose';
      SetLast(row);
      row := 'another source on the Network menu.';
      AddLine(row);
      exit(NetAsk)
   end;
   if rc <> 0 then begin
      big := concat('No time: ', ntWhy);
      Clip(big, row);
      SetLast(row);
      if (ntRc = fnNoRep) or (ntRc = fnBadFr) or (ntRc = fnStall) then begin

function NowUT (var dn : longint; var fr : real; var guess : boolean) : boolean;
{ Now, in UT: whole days since 1 January 2000 and the fraction of the   }
{ day.  The Lisa's clock gives the month, day and time (your time); the  }
{ real year comes from the twin-year note, or is guessed (the first year }
{ from baseYear on whose twin is the clock's year).  FALSE: clock unset. }
var y, doy, h, mi, sec : integer;
    ry, rm, rd, rt     : integer;
    m, d               : integer;
    realY, cand        : integer;
    exact, found       : boolean;
    haveN              : boolean;
    mins               : longint;
begin
   NowUT := false;
   guess := false;
   if not TwGet(y, doy, h, mi, sec) then
      exit(NowUT);
   TwMonDay(y, doy, m, d);
   haveN := TwLoad(myVol, ry, rm, rd, rt);
   if haveN and (rt = y) then
      realY := ry
   else if haveN and (y > rt) and (y - rt <= 2) then begin
      { the clock ran past New Year since it was set: 1987 -> 1988 is }
      { 2026 -> 2027, not the next year whose twin is 1988           }
      realY := ry + (y - rt);
      guess := true
   end
   else begin
      realY := y;
      guess := true;
      found := false;
      cand := baseYear;
      while (not found) and (cand <= baseYear + 40) do begin
         if TwYear(cand, exact) = y then begin
            realY := cand;
            found := true
         end
         else
            cand := cand + 1
      end
   end;
   dn := SaDays(realY, m, d);
   mins := h;
   mins := mins * 60 + mi - vwZoneQ * 15;
   while mins < 0 do begin
      mins := mins + 1440;
      dn := dn - 1
   end;
   while mins >= 1440 do begin
      mins := mins - 1440;
      dn := dn + 1
   end;
   fr := (mins + sec / 60.0) / 1440.0;
   pgDay := dn;
   pgFr := fr;
   NowUT := true
end;

function ParseNum (var t : StatStr; var x : real) : boolean;
{ "-84.51" -> -84.51.  Digits, at most one point, a minus sign first. }
var i, n    : integer;
    neg     : boolean;
    point   : boolean;
    scale   : real;
    got     : boolean;
    c       : char;
begin
   ParseNum := false;
   x := 0;
   neg := false;
   point := false;
   got := false;
   scale := 1.0;
   n := length(t);
   for i := 1 to n do begin
      c := t[i];
      if (c = '-') and (i = 1) then
         neg := true
      else if (c = '.') and (not point) then
         point := true
      else if (c >= '0') and (c <= '9') then begin
         got := true;
         if point then begin
            scale := scale / 10.0;
            x := x + (ord(c) - ord('0')) * scale
         end
         else
            x := x * 10.0 + (ord(c) - ord('0'))
      end
      else
         exit(ParseNum)
   end;
   if neg then x := -x;
   ParseNum := got
end;

procedure GeoDone;
{ RETURN at a LisaSolstice prompt: check, keep, save, show. }
var x   : real;
    ok  : boolean;
    row : StatStr;
begin
   ok := true;
   if length(promptTxt) = 0 then begin
      if promptStep = esUtc then x := vwZoneQ / 4.0
      else if promptStep = esLat then x := vwLat
      else x := vwLon
   end
   else
      ok := ParseNum(promptTxt, x);
   if ok then begin
      if promptStep = esUtc then ok := (x >= -12.0) and (x <= 14.0)
      else if promptStep = esLat then ok := (x >= -90.0) and (x <= 90.0)
      else ok := (x >= -180.0) and (x <= 180.0)
   end;
   if not ok then begin
      row := 'That is not a number in range.  Type it again, or X to cancel.';
      AddLine(row);
      GeoAsk;
      exit(GeoDone)
   end;
   promptOn := false;
   if promptStep = esLat then begin
      inLatT := x;
      promptStep := esLon;
      GeoAsk;
      exit(GeoDone)
   end;
   if promptStep = esLon then begin
      vwLat := inLatT;
      vwLon := x
   end
   else
      vwZoneQ := round(x * 4.0);
   promptStep := esNone;
   if not VwSave(myVol) then begin
      row := 'Kept for now, but SOLSTICE.DATA could not be written.';
      AddLine(row)
   end;
   DoTown
end;

procedure PromptKey (ch : char);
{ RETURN (or ENTER) finishes the value, Backspace erases, X or Clear    }
{ cancels, digits append up to numCap.  Everything else is ignored.     }
var tmp : string[1];
begin
   if (ch = chr(13)) or (ch = chr(3)) then begin
      if promptStep = esZone then
         ZoneDone
      else if promptStep >= esUtc then
         GeoDone
      else
         StepDone
   end
   else if (ch = chr(27)) or (((ch = 'X') or (ch = 'x')) and (promptStep <> esZone)) then
      CancelSet
   else if ch = chr(8) then begin
      if length(promptTxt) > 0 then begin
         promptTxt[0] := chr(length(promptTxt) - 1);
         DrawPrompt
      end
   end
   else if promptStep >= esUtc then begin
      if (((ch >= '0') and (ch <= '9')) or (ch = '-') or (ch = '.'))
         and (length(promptTxt) < 8) then begin
         tmp := ' ';
         tmp[1] := ch;
         promptTxt := concat(promptTxt, tmp);
         DrawPrompt
      end
   end
   else if promptStep = esZone then begin
      if (ch > ' ') and (ch <= '~') and (length(promptTxt) < ntZCap) then begin
         tmp := ' ';
         tmp[1] := ch;
         promptTxt := concat(promptTxt, tmp);
         DrawPrompt
      end
   end
   else if (ch >= '0') and (ch <= '9') and (length(promptTxt) < numCap)
   then begin
      tmp := ' ';
      tmp[1] := ch;
      promptTxt := concat(promptTxt, tmp);
      DrawPrompt
   end
end;

   case theEvent.what of

      buttonDown :
         if theEvent.who = menuFolder then begin
            MenuSelect(theEvent.where, theMenu, theItem);
            if (theMenu <> 0) and (theItem <> 0) then
               MenuCommand(theMenu, theItem);
            HiLiteMenu(0)
         end
         else if (theEvent.who = myWindow) and (pageMode = pgSplash) and (not promptOn) then
            DoDays;                             { a click on the splash goes on }

      buttonUp :
         ;

      keyDown :
         if (theEvent.who = myWindow) and (not theEvent.repeatKey) then begin
            if theEvent.appleKey then begin
               MenuKey(theEvent.ascii, theMenu, theItem);
               if theMenu <> 0 then begin
                  HiLiteMenu(theMenu);
                  MenuCommand(theMenu, theItem);
                  HiLiteMenu(0)
               end
            end
            else if promptOn then
               PromptKey(theEvent.ascii)
            else
               PlainKey(theEvent.ascii)
         end;

      folderActivate :
         begin
            TakeControl(theEvent, false, false);
            PutUpMenus
         end;

      folderDeactivate :
         GiveControl(theEvent);

      folderUpdate :
         if theEvent.who = myWindow then
            doUpdateEvent(true, theEvent.who)
         else begin
            BeginUpdate(theEvent.who);
            EndUpdate(theEvent.who)
         end;

      folderMoved :
         ;

      filerEvent :
         DoFilerEvent;

      abortEvent, diedEvent, diskEvent, catalogEvent, nilEvent :
         ;

      otherwise

   end
end;

{------------------------------------------------------------------------}
{ Initialize.                                                            }
{------------------------------------------------------------------------}

procedure Initialize;
var err      : integer;
    schedErr : integer;
    tmp1     : string[1];
    pcInfo   : procinfoRec;
    excName  : t_ex_name;
    myPrefix : TParamAlert;
    cutAt    : integer;
begin
   initing := true;

   excName := 'SYS_TERMINATE';
   Declare_Excep_Hdl(err, excName, @LastWishes);
   if err <> 0 then begin
      TellFiler(err, initFailed, noMemory, nil);
      halt
   end;

   toldToDie := false;
   myWindow  := nil;
   statCnt   := 0;
   quiet     := false;
   promptOn  := false;
   promptStep := esNone;
   promptTxt := '';
   promptLbl := '';
   inYear    := baseYear;
   inMonth   := 1;
   inDay     := 1;
   inHour    := 0;
   inMin     := 0;
   dfYear    := baseYear;
   dfMonth   := -1;
   dfDay     := -1;
   dfHour    := -1;
   dfMin     := -1;
   twEcode   := 0;
   ckMode    := 0;
   pageMode  := pgText;
   factNo    := 0;
   phNo      := 1;
   inUtcQ    := 0;
   inLatT    := 0;
   pgDay     := 0;
   pgFr      := 0;
   SaInit;
   VwInit;

   InitHeap(@myZone[0], @myZone[maxHeap], @OutOfHeapSpace);

   Sched_Class(schedErr, false);
   Info_Process(err, My_ID, pcInfo);
   Sched_Class(schedErr, true);

   cutAt := pos('}', pcInfo.progpathname);
   if (cutAt = 0) or (cutAt > 40) then begin
      TellFiler(err, initFailed, internalError, nil);
      halt
   end;
   myPrefix := copy(pcInfo.progpathname, 1, cutAt);
   { the device prefix alone: everything before the open-brace of the  }
   { tool file set.  chr(123) is the open brace, built the SanName way. }
   tmp1 := ' ';
   tmp1[1] := chr(123);
   cutAt := pos(tmp1, pcInfo.progpathname);
   if (cutAt > 1) and (cutAt <= 33) then
      myVol := copy(pcInfo.progpathname, 1, cutAt - 1)
   else
      myVol := '-#12-';                { defensive fallback: boot }
   NtInit;                              { REV D: the network source (calls FnInit) }
   if NtLoad(myVol) then ;
   if NtTownLd(myVol) then ;
   FnSetVol(myVol);                     { REV J 1-Nov-2026: AFTER NtInit, whose FnInit had wiped }
   if FnMbUp then ;                     { the ESProFile probe (2/10: N fell back to Serial B) }
   if VwLoad(myVol) then ;              { LisaSolstice: your town and time zone }

   InitMP(theHeap, myPrefix, @AlertFailed, alertVerNum,
          @myMenus[mFile], allAlerts, err);
   if err > 0 then begin
      TellFiler(err, initFailed, noMemory, nil);
      halt
   end;

   InitWmlCrs(err);
   if err <> 0 then begin
      TellFiler(err, initFailed, noDiskSpace, nil);
      halt
   end;

   SetPort(deskPort);

   initing := false
end;

{------------------------------------------------------------------------}
{ The main program -- two-phase loop, nothing to do when idle.           }
{------------------------------------------------------------------------}

begin { main program }

   InitGraf(@thePort);
   OpenWm;
   Initialize;

   repeat

      while not (ImActive or toldToDie) do begin
         GetEvent(theEvent);
         ProcessTheEvent
      end;

      while ImActive and (not toldToDie) do begin
         if EventAvail then begin
            GetEvent(theEvent);
            ProcessTheEvent
         end
         else
            LetOthersRun
      end;

   until toldToDie;

   termErr := 0
end.
