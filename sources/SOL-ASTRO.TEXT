unit SolAstro;

{ SOL-ASTRO.TEXT -- LisaSolstice (tool 419326): the astronomy.             }
{ REV A  2-Nov-2026  Equinoxes and solstices (Meeus, Astronomical         }
{                    Algorithms, ch. 27, years 2000-3000, all 24 periodic  }
{                    terms), the place of the Sun (Meeus ch. 25, low       }
{                    accuracy: about 0.01 degree), sunrise and sunset      }
{                    (the -0.833 degree standard), and the calendar.       }
{ Lisa Pascal reals are single precision, so a moment is kept as whole     }
{ days since 1 January 2000, 0:00 UT, in a longint, plus a real fraction   }
{ of a day; every big constant is split into its whole and its fraction.   }
{ About a minute of accuracy for the four events (Delta T is a fixed       }
{ estimate).  No other unit is needed.                                     }

INTERFACE

const
   saPi = 3.14159265;

var
   saDecl  : real;      { declination of the Sun, degrees, north +  (SaSun) }
   saLam   : real;      { ecliptic longitude of the Sun, degrees 0..360     }
   saEarth : real;      { that of the Earth, seen from the Sun (saLam + 180) }
   saEoT   : real;      { equation of time, minutes (sundial - clock)       }
   saSubLt : real;      { where the Sun is overhead: latitude, north +      }
   saSubLn : real;      { and longitude, east +, -180..180                  }

procedure SaInit;
function  SaDays  (y, m, d : integer) : longint;
procedure SaCiv   (dayN : longint; var y, m, d : integer);
function  SaWkDay (dayN : longint) : integer;
procedure SaEvent (year, k : integer; var dayN : longint; var fr : real);
procedure SaSun   (dayN : longint; fr : real);
function  SaRise  (dayN : longint; lat, lon : real; var rise, sett : real) : integer;
function  SaMod   (x, m : real) : real;

IMPLEMENTATION

{$S SOLAstro}

var
   saTA : array [1..24] of real;
   saTB : array [1..24] of real;
   saTC : array [1..24] of real;

procedure SaT3 (i : integer; a, b, c : real);
begin
   saTA[i] := a;
   saTB[i] := b;
   saTC[i] := c
end;

function SaMod {x, m : real} : real;
{ x reduced to 0 <= x < m. }
var r : real;
begin
   r := x - m * trunc(x / m);
   if r < 0 then r := r + m;
   SaMod := r
end;

function SaRad (deg : real) : real;
begin
   SaRad := deg * saPi / 180.0
end;

function SaDeg (rad : real) : real;
begin
   SaDeg := rad * 180.0 / saPi
end;

function SaAt2 (y, x : real) : real;
{ The angle of (x, y), radians, -pi..pi. }
var r : real;
begin
   if x > 0 then
      r := arctan(y / x)
   else if x < 0 then begin
      if y >= 0 then r := arctan(y / x) + saPi else r := arctan(y / x) - saPi
   end
   else if y > 0 then
      r := saPi / 2.0
   else if y < 0 then
      r := -saPi / 2.0
   else
      r := 0;
   SaAt2 := r
end;

function SaAsin (x : real) : real;
begin
   if x >= 1.0 then SaAsin := saPi / 2.0
   else if x <= -1.0 then SaAsin := -saPi / 2.0
   else SaAsin := arctan(x / sqrt(1.0 - x * x))
end;

function SaDays {y, m, d : integer} : longint;
{ Days from 1 January 2000 to y-m-d (proleptic Gregorian). }
var yy, era, yoe, mm, doy, doe : longint;
begin
   yy := y;
   if m <= 2 then yy := yy - 1;
   if yy >= 0 then era := yy div 400 else era := (yy - 399) div 400;
   yoe := yy - era * 400;
   if m > 2 then mm := m - 3 else mm := m + 9;
   doy := (153 * mm + 2) div 5 + d - 1;
   doe := yoe * 365 + yoe div 4 - yoe div 100 + doy;
   SaDays := era * 146097 + doe - 730425
end;

procedure SaCiv {dayN : longint; var y, m, d : integer};
var z, era, doe, yoe, yy, doy, mp, mm, dd : longint;
begin
   z := dayN + 730425;
   if z >= 0 then era := z div 146097 else era := (z - 146096) div 146097;
   doe := z - era * 146097;
   yoe := (doe - doe div 1460 + doe div 36524 - doe div 146096) div 365;
   yy := yoe + era * 400;
   doy := doe - (365 * yoe + yoe div 4 - yoe div 100);
   mp := (5 * doy + 2) div 153;
   dd := doy - (153 * mp + 2) div 5 + 1;
   if mp < 10 then mm := mp + 3 else mm := mp - 9;
   if mm <= 2 then yy := yy + 1;
   d := dd;
   m := mm;
   y := yy
end;

function SaWkDay {dayN : longint} : integer;
{ 0 = Sunday .. 6 = Saturday (1 January 2000 was a Saturday). }
var w : longint;
begin
   w := (dayN + 6) mod 7;
   if w < 0 then w := w + 7;
   SaWkDay := w
end;

procedure SaEvent {year, k : integer; var dayN : longint; var fr : real};
{ Moment (UT) of k = 0 the March equinox, 1 the June solstice, 2 the      }
{ September equinox, 3 the December solstice of the year: whole days      }
{ since 1 January 2000 0:00 UT in dayN, the fraction of the day in fr.    }
var n              : longint;
    yr, y2, y3, y4 : real;
    t, w, dl, s    : real;
    i              : integer;
    whole, part    : real;
    dlt            : real;
begin
   n := year - 2000;
   yr := n / 1000.0;
   y2 := yr * yr;
   y3 := y2 * yr;
   y4 := y3 * yr;
   { JDE0 - 2451544.5 (Meeus table 27.B), the whole days of the       }
   { 365242.xxxxx Y term apart: 365242.xxxxx Y = n * 365 + n * 0.2xxxx }
   case k of
      0: begin
            dayN := 79 + n * 365;
            part := 0.30984 + n * 0.24237404
                    + 0.05169 * y2 - 0.00411 * y3 - 0.00057 * y4
         end;
      1: begin
            dayN := 172 + n * 365;
            part := 0.06767 + n * 0.24162603
                    + 0.00325 * y2 + 0.00888 * y3 - 0.00030 * y4
         end;
      2: begin
            dayN := 265 + n * 365;
            part := 0.71715 + n * 0.24201767
                    - 0.11575 * y2 + 0.00337 * y3 + 0.00078 * y4
         end;
      3: begin
            dayN := 355 + n * 365;
            part := 0.55952 + n * 0.24274049
                    - 0.06223 * y2 - 0.00823 * y3 + 0.00032 * y4
         end
   end;
   t := (dayN + part - 0.5) / 36525.0;
   w := SaRad(SaMod(35999.373 * t - 2.47, 360.0));
   dl := 1.0 + 0.0334 * cos(w) + 0.0007 * cos(2.0 * w);
   s := 0;
   for i := 1 to 24 do
      s := s + saTA[i] * cos(SaRad(SaMod(saTB[i] + saTC[i] * t, 360.0)));
   part := part + 0.00001 * s / dl;
   { Delta T (TT - UT), seconds: about 69 s in 2026 }
   dlt := 69.0 + 0.4 * (year - 2026);
   part := part - dlt / 86400.0;
   whole := trunc(part);
   if part < whole then whole := whole - 1;
   dayN := dayN + trunc(whole);
   fr := part - whole
end;

procedure SaSun {dayN : longint; fr : real};
{ The Sun at that moment (UT): fills saDecl, saLam, saEarth, saEoT,      }
{ saSubLt, saSubLn.                                                       }
var nd           : real;
    l, g, lam, e : real;
    ra, ut       : real;
begin
   nd := (dayN - 0.5) + fr;                    { days since J2000.0 }
   l := SaMod(280.460 + SaMod(0.9856474 * dayN, 360.0) + 0.9856474 * (fr - 0.5), 360.0);
   g := SaMod(357.528 + SaMod(0.9856003 * dayN, 360.0) + 0.9856003 * (fr - 0.5), 360.0);
   lam := SaMod(l + 1.915 * sin(SaRad(g)) + 0.020 * sin(SaRad(2.0 * g)), 360.0);
   e := 23.439 - 0.0000004 * nd;
   saLam := lam;
   saEarth := SaMod(lam + 180.0, 360.0);
   saDecl := SaDeg(SaAsin(sin(SaRad(e)) * sin(SaRad(lam))));
   ra := SaMod(SaDeg(SaAt2(cos(SaRad(e)) * sin(SaRad(lam)), cos(SaRad(lam)))), 360.0);
   saEoT := l - ra;
   if saEoT > 180.0 then saEoT := saEoT - 360.0;
   if saEoT < -180.0 then saEoT := saEoT + 360.0;
   saEoT := saEoT * 4.0;                        { degrees of time -> minutes }
   saSubLt := saDecl;
   ut := fr * 24.0;
   saSubLn := -15.0 * (ut - 12.0 + saEoT / 60.0);
   saSubLn := SaMod(saSubLn + 180.0, 360.0) - 180.0
end;

function SaRise {dayN : longint; lat, lon : real; var rise, sett : real} : integer;
{ Sunrise and sunset (UT hours, may be < 0 or >= 24) on that day for a    }
{ place at lat (north +), lon (east +).  0 = both happen, 1 = the Sun    }
{ stays down all day, 2 = it stays up all day.                            }
var noon, c, h : real;
begin
   SaSun(dayN, 0.5 - lon / 360.0);              { near local noon }
   noon := 12.0 - lon / 15.0 - saEoT / 60.0;
   c := (sin(SaRad(-0.833)) - sin(SaRad(lat)) * sin(SaRad(saDecl)))
        / (cos(SaRad(lat)) * cos(SaRad(saDecl)));
   rise := noon;
   sett := noon;
   if c >= 1.0 then
      SaRise := 1
   else if c <= -1.0 then
      SaRise := 2
   else begin
      h := SaDeg(saPi / 2.0 - SaAsin(c)) / 15.0;   { acos(c), in hours }
      rise := noon - h;
      sett := noon + h;
      SaRise := 0
   end
end;
